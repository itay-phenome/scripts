          "#!/bin/bash",
          "LOGFILE=\"/tmp/check_cyservice.log\"",
          "SERVICE=\"cyservice\"",
          "INSTALL_DIR=\"/home/ubuntu/files/CyNet\"",
          "S3_ZIP=\"s3://phenome-devops-files/cynet_installation/LINUX_CynetEPSInstaller.zip\"",
          "ZIP_NAME=\"LINUX_CynetEPSInstaller.zip\"",
          "ISRAEL_TIME=$(TZ=\"Asia/Jerusalem\" date +\"%Y-%m-%d %H:%M:%S %Z\")",
          "echo \"$ISRAEL_TIME: Starting Cynet check and install procedure\" | tee -a \"$LOGFILE\"",
          "REQUIRED_CMDS=(systemctl aws unzip mkdir touch chmod)",
          "MISSING=0",

          "  if ! command -v \"$CMD\" >/dev/null 2>&1; then",
          "    echo \"$ISRAEL_TIME: ❌ Required command '$CMD' is missing.\" | tee -a \"$LOGFILE\"",
          "    MISSING=1",
          "  fi",
          "done",
          "if [ \"$MISSING\" -eq 1 ]; then",
          "  echo \"$ISRAEL_TIME: ❌ One or more required commands are missing. Aborting.\" | tee -a \"$LOGFILE\"",

          "fi",
          "if [ ! -f \"$LOGFILE\" ]; then",
          "  sudo mkdir -p \"$(dirname \\\"$LOGFILE\\\")\" || exit 1",

          "fi",
          "if systemctl list-unit-files | grep -q \"^${SERVICE}.service\"; then",
          "  echo \"$ISRAEL_TIME: $SERVICE service already exists\" | tee -a \"$LOGFILE\"",
          "else",
          "  echo \"$ISRAEL_TIME: $SERVICE service not found, proceeding with installation\" | tee -a \"$LOGFILE\"",
          "  if [ ! -d \"$INSTALL_DIR\" ]; then",
          "    sudo mkdir -p \"$INSTALL_DIR\" || { echo \"$ISRAEL_TIME: Failed to create $INSTALL_DIR\" | tee -a \"$LOGFILE\"; exit 1; }",
          "  fi",
          "  echo \"$ISRAEL_TIME: Downloading installer from $S3_ZIP\" | tee -a \"$LOGFILE\"",
          "  sudo aws s3 cp \"$S3_ZIP\" \"$INSTALL_DIR/$ZIP_NAME\" || {",
          "    echo \"$ISRAEL_TIME: ❌ Failed to download from S3\" | tee -a \"$LOGFILE\"",
          "    exit 1",
          "  }",
          "  cd \"$INSTALL_DIR\" || { echo \"$ISRAEL_TIME: ❌ Failed to cd to $INSTALL_DIR\" | tee -a \"$LOGFILE\"; exit 1; }",
          "  sudo unzip -o \"$ZIP_NAME\" || {",
          "    echo \"$ISRAEL_TIME: ❌ Failed to unzip $ZIP_NAME\" | tee -a \"$LOGFILE\"",
          "    exit 1",
          "  }",
          "  sudo chmod +x *.sh || { echo \"$ISRAEL_TIME: ❌ Failed to chmod .sh files\" | tee -a \"$LOGFILE\"; exit 1; }",
          "  if [ -x \"./cy_easyinstall.sh\" ]; then",
          "    echo \"$ISRAEL_TIME: Running installer\" | tee -a \"$LOGFILE\"",
          "    sudo ./cy_easyinstall.sh || { echo \"$ISRAEL_TIME: ❌ Installer failed\" | tee -a \"$LOGFILE\"; exit 1; }",
          "    sleep 5",
          "  else",
          "    echo \"$ISRAEL_TIME: ❌ cy_easyinstall.sh not found or not executable\" | tee -a \"$LOGFILE\"",
          "    exit 1",
          "  fi",
          "fi",
          "if systemctl list-unit-files | grep -q \"^${SERVICE}.service\"; then",
          "  if systemctl is-active --quiet \"$SERVICE\"; then",
          "    echo \"$ISRAEL_TIME: ✅ $SERVICE is installed and running\" | tee -a \"$LOGFILE\"",
          "  else",
          "    echo \"$ISRAEL_TIME: ⚠️ $SERVICE is installed but not active\" | tee -a \"$LOGFILE\"",
          "  fi",
          "else",
          "  echo \"$ISRAEL_TIME: ❌ $SERVICE is still not installed after attempt\" | tee -a \"$LOGFILE\"",
          "  exit 1",
          "fi"
